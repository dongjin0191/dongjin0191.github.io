📋 주문 모으기 (총무용)

② script.new 열기
  1. 크롬 메뉴(⋮) → 데스크톱 사이트 체크
  2. 편집기 글자 아무 데나 한 번 누르고 모두 선택(길게 누르기 → 전체 선택) → 지우기 → 붙여넣기 → 💾 저장
    코드는 한 줄로 되어 있어 폰에서 붙여도 깨지지 않습니다. 줄 수가 1줄이면 정상.
  3. 배포 → 새 배포 → 톱니 → 웹 앱 → 액세스 모든 사용자 → 배포
  4. 권한 승인: 고급 → (안전하지 않음)으로 이동 → 허용
  5. 나온 웹 앱 URL(…/exec)을 복사해 Claude에게 보내기
var SHEET='주문';function book_(){var ss=null;try{ss=SpreadsheetApp.getActive();}catch(e){}if(ss)return ss;var pr=PropertiesService.getScriptProperties(),id=pr.getProperty('SHEET_ID');if(id){try{return SpreadsheetApp.openById(id);}catch(e){}}ss=SpreadsheetApp.create('제주수양회 주문');pr.setProperty('SHEET_ID',ss.getId());return ss;}function sheet_(){var ss=book_();var sh=ss.getSheetByName(SHEET)||ss.insertSheet(SHEET);if(sh.getLastRow()===0)sh.appendRow(['시각','장소','이름','메뉴','선택','메모']);return sh;}function json_(o){return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON);}function rows_(){var v=sheet_().getDataRange().getValues();v.shift();return v.map(function(r){return {at:r[0],place:r[1],name:r[2],menu:r[3],opt:r[4],memo:r[5]};});}function doGet(){return json_({ok:true,orders:rows_()});}function doPost(e){var lock=LockService.getScriptLock();lock.waitLock(10000);try{var d=JSON.parse(e.postData.contents||'{}');var place=String(d.place||'').slice(0,40),name=String(d.name||'').slice(0,20);if(!place||!name)return json_({ok:false,error:'need place and name'});var sh=sheet_(),v=sh.getDataRange().getValues();for(var i=v.length-1;i>=1;i--)if(v[i][1]===place&&v[i][2]===name)sh.deleteRow(i+1);if(d.menu)sh.appendRow([new Date(),place,name,String(d.menu).slice(0,40),String(d.opt||'').slice(0,40),String(d.memo||'').slice(0,100)]);return json_({ok:true,orders:rows_()});}finally{lock.releaseLock();}}
